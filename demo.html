ES Module Usage

import { ForgeExprEvaluatorUtil } from './dist/browser/forge-expr-evaluator.es.js';

const datum = { /* your datum data */ };
const sourceCode = `
#lang forge
sig Node { edges: set Node }
pred connected { all n: Node | some n.edges }
`;

const evaluator = new ForgeExprEvaluatorUtil(datum, sourceCode);
const result = evaluator.evaluateExpression('Node');
console.log(result);
